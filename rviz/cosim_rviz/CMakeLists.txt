cmake_minimum_required(VERSION 3.16)
project(cosim_rviz)
set(CMAKE_AUTOMOC ON)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(ground_panel SHARED src/ground_panel.cpp include/cosim_rviz/ground_panel.hpp)
target_include_directories(ground_panel PUBLIC include)
target_link_libraries(ground_panel Qt5::Widgets)
ament_target_dependencies(ground_panel rviz_common pluginlib rclcpp std_msgs)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS ground_panel LIBRARY DESTINATION lib)
ament_package()

if(BUILD_TESTING)
  add_executable(panel_check tests/panel_check.cpp)
  target_link_libraries(panel_check Qt5::Widgets)
  ament_target_dependencies(panel_check rviz_common pluginlib)
endif()
